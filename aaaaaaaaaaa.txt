  <script>
    // ============================================================
    //  📋 СПИСОК КАНАЛОВ — РЕДАКТИРУЙТЕ ЗДЕСЬ!
    //
    //  Формат:
    //    name  — название канала
    //    desc  — краткое описание
    //    url   — ссылка на HLS-поток (.m3u8)
    //
    //  Где взять потоки:
    //    • https://github.com/iptv-org/iptv — тысячи бесплатных
    //      каналов со всего мира (ищите ссылки .m3u8)
    //    • Свой сервер (OBS + nginx-rtmp, например)
    // ============================================================
    const channels = [
      {
        name: "🎬 Big Buck Bunny",
        desc: "Тестовый поток (демо)",
        url: "https://test-streams.mux.dev/x36xhzz/x36xhzz.m3u8"
      },
      {
        name: "🌍 France 24 (EN)",
        desc: "Новости на английском",
        url: "https://stream.france24.com/live/hls/f24_en_hi.m3u8"
      },
      {
        name: "🎵 Lofi Hip Hop",
        desc: "Музыка 24/7",
        url: "https://play.streamafrica.net/lofiradio"
      },
      {
        name: "🔬 NASA TV",
        desc: "Космос и наука",
        url: "https://ntv1.akamaized.net/hls/live/2014075/NASA-NTV1-HLS/master_2000.m3u8"
      },
      {
        name: "📰 DW News",
        desc: "Немецкие новости (EN)",
        url: "https://dwamdstream102.akamaized.net/hls/live/2015525/dwstream102/index.m3u8"
      }
    ];

    // ============================================================
    //  ДВИЖОК ПЛЕЕРА — НЕ ТРОГАЙТЕ, ЕСЛИ НЕ ЗНАЕТЕ, ЧТО ДЕЛАЕТЕ
    // ============================================================
    const video = document.getElementById('video-player');
    const placeholder = document.getElementById('placeholder');
    const channelListEl = document.getElementById('channel-list');
    const nowPlayingText = document.getElementById('now-playing-text');
    const statusDot = document.getElementById('status-dot');
    let hls = null;
    let activeIndex = -1;

    // Генерация списка каналов
    channels.forEach((ch, i) => {
      const div = document.createElement('div');
      div.className = 'channel-item';
      div.innerHTML = `
        <div class="channel-number">${i + 1}</div>
        <div class="channel-info">
          <h3>${ch.name}</h3>
          <p>${ch.desc}</p>
        </div>
      `;
      div.addEventListener('click', () => playChannel(i));
      channelListEl.appendChild(div);
    });

    // Воспроизведение канала
    function playChannel(index) {
      const ch = channels[index];
      activeIndex = index;

      // Обновляем активный класс
      document.querySelectorAll('.channel-item').forEach((el, i) => {
        el.classList.toggle('active', i === index);
      });

      // Обновляем инфо-панель
      nowPlayingText.textContent = `${ch.name} — ${ch.desc}`;
      statusDot.style.display = 'block';
      placeholder.style.display = 'none';

      // Уничтожаем предыдущий HLS-инстанс
      if (hls) {
        hls.destroy();
        hls = null;
      }

      const url = ch.url;

      if (Hls.isSupported()) {
        hls = new Hls({
          enableWorker: true,
          lowLatencyMode: true
        });
        hls.loadSource(url);
        hls.attachMedia(video);
        hls.on(Hls.Events.MANIFEST_PARSED, () => {
          video.play().catch(() => {});
        });
        hls.on(Hls.Events.ERROR, (event, data) => {
          if (data.fatal) {
            nowPlayingText.textContent = `⚠ Ошибка: ${ch.name} недоступен`;
            statusDot.style.display = 'none';
          }
        });
      } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
        // Safari
        video.src = url;
        video.addEventListener('loadedmetadata', () => {
          video.play().catch(() => {});
        });
      } else {
        nowPlayingText.textContent = '⚠ Ваш браузер не поддерживает HLS';
      }
    }

    // Полный экран
    function toggleFullscreen() {
      if (!document.fullscreenElement) {
        video.requestFullscreen().catch(() => {});
      } else {
        document.exitFullscreen();
      }
    }

    // Горячие клавиши
    document.addEventListener('keydown', (e) => {
      if (e.key === 'ArrowUp' && activeIndex > 0) {
        playChannel(activeIndex - 1);
      } else if (e.key === 'ArrowDown' && activeIndex < channels.length - 1) {
        playChannel(activeIndex + 1);
      } else if (e.key === 'f' || e.key === 'F') {
        toggleFullscreen();
      }
    });
  </script>
